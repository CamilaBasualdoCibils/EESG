cmake_minimum_required(VERSION 3.14)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND DEFINED ENV{VCPKG_ROOT})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "Toolchain")
endif()
project(EESG VERSION 0.1.0 LANGUAGES C CXX)
option(EESG_BUILD_TESTING "Build EESG tests" OFF)
set(LIGHTHOUSE_EXTENDED_BUILD_EXAMPLES ON)
if(NOT TARGET Lighthouse::Extended)
    add_subdirectory(extern/Lighthouse-Extended)
endif()
find_package(Threads REQUIRED)
add_library(EESGWorld STATIC src/World/World.cpp src/World/Terrain/TerrainSystem.cpp)
target_include_directories(EESGWorld PUBLIC src)
target_link_libraries(EESGWorld PUBLIC Lighthouse::Extended)
set_source_files_properties(src/World/Terrain/TerrainSystem.cpp PROPERTIES COMPILE_DEFINITIONS EESG_TERRAIN_RESOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/res/terrain")
add_executable(EESG src/Main.cpp src/GameManager.cpp)
target_link_libraries(EESG PRIVATE EESGWorld Threads::Threads Lighthouse::GLFWInputBackend Lighthouse::OpenGLBackend Lighthouse::VulkanBackend)
target_compile_definitions(EESG PRIVATE EESG_RESOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/res" EESG_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
if(EESG_BUILD_TESTING)
    enable_testing()
    add_executable(EESGGameAssetTests tests/GameAssetTests.cpp)
    target_link_libraries(EESGGameAssetTests PRIVATE Lighthouse::Extended)
    target_compile_definitions(EESGGameAssetTests PRIVATE EESG_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME EESGGameAssetTests COMMAND EESGGameAssetTests)
    add_executable(EESGOctreeTests tests/AdaptiveOctreeTests.cpp)
    target_include_directories(EESGOctreeTests PRIVATE src/Types)
    target_compile_features(EESGOctreeTests PRIVATE cxx_std_23)
    add_test(NAME EESGOctreeTests COMMAND EESGOctreeTests)
    add_executable(EESGTerrainResidencyTests tests/TerrainResidencyTests.cpp)
    target_link_libraries(EESGTerrainResidencyTests PRIVATE EESGWorld Lighthouse::OpenGLBackend Lighthouse::VulkanBackend)
    add_test(NAME EESGTerrainResidencyTests COMMAND EESGTerrainResidencyTests)
endif()
